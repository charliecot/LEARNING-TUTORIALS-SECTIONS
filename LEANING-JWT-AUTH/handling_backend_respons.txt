from rest_framework.response import Response
from rest_framework import status

"""
========================================================
📦 STANDARD API RESPONSE HELPERS
========================================================

Why we use this:
- Keeps all API responses consistent
- Easier frontend handling
- Cleaner and professional backend structure
========================================================
"""

# ✅ SUCCESS RESPONSE
def success_response(message, data=None, status_code=status.HTTP_200_OK):
    """
    Use this when request is successful.

    Parameters:
        message (str) → human readable success message
        data (dict/any) → actual response data
        status_code → HTTP status code (default 200)
    """

    return Response({
        "status": "success",
        "message": message,
        "data": data
    }, status=status_code)


# ❌ ERROR RESPONSE
def error_response(message, status_code=status.HTTP_400_BAD_REQUEST):
    """
    Use this when something goes wrong.

    Parameters:
        message (str or dict) → error message or serializer errors
        status_code → HTTP error status code
    """

    return Response({
        "status": "error",
        "message": message,
        "data": None
    }, status=status_code)
