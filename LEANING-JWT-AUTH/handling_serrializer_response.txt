    password = serializers.CharField(write_only=True)

    # 🔍 FIELD VALIDATION (email)
    def validate_email(self, value):
        """
        Runs only for email field

# ======================================================
# 🔥 CUSTOM REUSABLE VALIDATOR FUNCTION
# ======================================================
def no_spaces_validator(value):
    """
    🚫 Reject values that contain spaces
    Reusable in multiple fields
    """
    if " " in value:
        raise serializers.ValidationError("Spaces are not allowed")
    return value


# ======================================================
# 🧾 MAIN SERIALIZER EXAMPLE
# ======================================================
class UserSerializer(serializers.Serializer):

    # ==================================================
    # 📌 BASIC FIELDS
    # ==================================================
    email = serializers.EmailField()
    password = serializers.CharField(write_only=True)
    username = serializers.CharField()

    age = serializers.IntegerField()

    phone = serializers.CharField(
        validators=[
            RegexValidator(
                regex=r'^\d{10}$',
                message="Phone must be exactly 10 digits"
            )
        ]
    )

    status = serializers.ChoiceField(
        choices=(
            ("active", "Active"),
            ("inactive", "Inactive"),
        )
    )

    # ==================================================
    # 🔐 FIELD-LEVEL VALIDATION (single field)
    # ==================================================

    def validate_email(self, value):
        """
        Runs ONLY for email field
        """
        if not value.endswith("@gmail.com"):
            raise serializers.ValidationError("Only Gmail allowed")

        return value

    def validate_password(self, value):
        """
        Password strength check
        """
        if len(value) < 6:
            raise serializers.ValidationError("Password too short")

        if value.isnumeric():
            raise serializers.ValidationError("Password cannot be only numbers")

        return value

    def validate_username(self, value):
        """
        Reusable validator example
        """
        return no_spaces_validator(value)

    def validate_age(self, value):
        """
        Age validation
        """
        if value < 18:
            raise serializers.ValidationError("Must be 18+")

        return value

    # ==================================================
    # 🔥 CROSS-FIELD VALIDATION (important)
    # ==================================================
    def validate(self, data):
        """
        Runs after all field validations
        Used for comparing multiple fields
        """

        email = data.get("email")
        password = data.get("password")

        if email == password:
            raise serializers.ValidationError(
                "Email and password cannot be the same"
            )

        return data
