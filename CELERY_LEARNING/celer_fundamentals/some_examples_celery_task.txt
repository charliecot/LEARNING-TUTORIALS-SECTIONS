from celery import shared_task, current_task
from django.core.mail import send_mail
from django.contrib.auth.models import User
import time


# =========================================================
# 1. BASIC SHARED TASK (simple function)
# =========================================================

@shared_task
def add(x, y):
    """
    Simple task: adds two numbers
    """
    return x + y


# =========================================================
# 2. TASK WITH DELAY SIMULATION
# =========================================================

@shared_task
def long_running_task():
    """
    Simulates a heavy task (e.g. report generation)
    """
    time.sleep(10)
    return "Task completed"


# =========================================================
# 3. TASK WITH EMAIL SENDING
# =========================================================

@shared_task
def send_welcome_email(user_email):
    """
    Sends welcome email after user registration
    """
    send_mail(
        subject="Welcome!",
        message="Thanks for joining our platform",
        from_email="admin@example.com",
        recipient_list=[user_email],
        fail_silently=False,
    )


# =========================================================
# 4. TASK WITH COUNTDOWN (RUN LATER)
# =========================================================

@shared_task
def reminder_task(message):
    """
    Example of delayed execution using countdown
    """
    return f"Reminder: {message}"


# =========================================================
# 5. RETRY TASK (VERY IMPORTANT IN REAL PROJECTS)
# =========================================================

@shared_task(bind=True, max_retries=3)
def fragile_task(self):
    """
    Task that may fail and retries automatically
    """
    try:
        # simulate failure
        raise ValueError("Something went wrong")
    except Exception as exc:
        raise self.retry(exc=exc, countdown=5)


# =========================================================
# 6. TASK WITH CURRENT TASK INFO
# =========================================================

@shared_task(bind=True)
def progress_task(self):
    """
    Shows how to access task metadata
    """
    for i in range(1, 6):
        time.sleep(1)
        print(f"Step {i}/5")

    return "Done"


# =========================================================
# 7. IGNORE RESULT TASK (faster performance)
# =========================================================

@shared_task(ignore_result=True)
def log_event(message):
    """
    Used for logging (no result stored in backend)
    """
    print(f"LOG: {message}")
