db:                          # Name of the service/container inside docker-compose. Django will use this name as HOST (db).

  image: postgres:16         # Pulls PostgreSQL version 16 from Docker Hub.

  container_name: postgres_container   # Custom container name shown in `docker ps`.

  restart: always            # Automatically restarts container if it stops or Docker restarts.

  environment:               # Environment variables used to configure PostgreSQL on first startup.

    POSTGRES_DB: mydb        # Creates a database named "mydb".

    POSTGRES_USER: user      # Creates a PostgreSQL user named "user".

    POSTGRES_PASSWORD: userpassword   # Password for that PostgreSQL user.

  ports:                     # Maps ports between your PC and container.

    - "5432:5432"           # Left side = your computer port, right side = container port.
                             # You connect from your machine using localhost:5432.

  volumes:                   # Persistent storage so database data remains after container stops.

    - postgres_data:/var/lib/postgresql/data
                             # postgres_data = Docker volume name
                             # /var/lib/postgresql/data = PostgreSQL internal data folder
