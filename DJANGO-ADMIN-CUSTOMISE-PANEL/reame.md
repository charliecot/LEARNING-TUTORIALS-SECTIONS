# MIGRATIONS
# ----------------------------------------------------------

# Create migrations for one app

# Create an empty migration
python manage.py makemigrations --empty app_name

# Create a named migration
python manage.py makemigrations app_name --name migration_name

# Apply all migrations
python manage.py migrate

# Apply migrations for one app
python manage.py migrate app_name

# Apply up to a specific migration
python manage.py migrate app_name 0003

# Fake a migration
python manage.py migrate --fake

# Fake initial migration
python manage.py migrate --fake-initial

# Show migrations
python manage.py showmigrations

# Show migrations for one app
python manage.py showmigrations app_name

# Show SQL generated by a migration
python manage.py sqlmigrate app_name 0001
